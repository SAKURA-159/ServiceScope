cmake_minimum_required(VERSION 3.16)
project(ServiceScope VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Fetch header-only dependencies
include(FetchContent)

FetchContent_Declare(
    cpp-httplib
    GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git
    GIT_TAG v0.18.1
)
FetchContent_Declare(
    json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
)
FetchContent_MakeAvailable(cpp-httplib json)

# Embed dashboard HTML at build time
set(GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")

file(READ "${CMAKE_SOURCE_DIR}/web/dashboard.html" DASHBOARD_HTML_CONTENT)
file(WRITE "${GENERATED_DIR}/dashboard_html.h"
"#pragma once
// Auto-generated by CMake — do not edit
const char* DASHBOARD_HTML = R\"rawliteral(${DASHBOARD_HTML_CONTENT})rawliteral\";
")

add_executable(servicescope
    src/main.cpp
)

target_include_directories(servicescope PRIVATE
    ${cpp-httplib_SOURCE_DIR}
    ${json_SOURCE_DIR}/include
    src
    ${GENERATED_DIR}
)

# ---- Unit Tests ----
option(BUILD_TESTS "Build unit tests" OFF)
if(BUILD_TESTS)
    add_executable(servicescope_tests
        tests/test_main.cpp
    )
    target_include_directories(servicescope_tests PRIVATE
        ${cpp-httplib_SOURCE_DIR}
        ${json_SOURCE_DIR}/include
        src
    )
    if(MSVC)
        target_compile_options(servicescope_tests PRIVATE /W4 /O2)
    else()
        target_compile_options(servicescope_tests PRIVATE -Wall -Wextra -O2)
        target_link_libraries(servicescope_tests PRIVATE pthread)
    endif()
endif()

# Platform-specific settings
if(WIN32)
    target_compile_definitions(servicescope PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(servicescope PRIVATE ws2_32)
endif()

if(MSVC)
    target_compile_options(servicescope PRIVATE /W4 /O2)
else()
    target_compile_options(servicescope PRIVATE -Wall -Wextra -O3 -march=native)
    target_link_libraries(servicescope PRIVATE pthread)
endif()
